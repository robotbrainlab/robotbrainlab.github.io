\section{Resolving a Conflict}\label{B-git-cheat-sheet:resolving-a-conflict}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \texttt{git\ status} lists files marked \texttt{both\ modified}.
\item
  Edit each file to its final form and delete the \texttt{\textless{}\textless{}\textless{}\textless{}\textless{}\textless{}\textless{}}, \texttt{=======}, and \texttt{\textgreater{}\textgreater{}\textgreater{}\textgreater{}\textgreater{}\textgreater{}\textgreater{}} lines.
\item
  Run the code to check it works, then \texttt{git\ add} each file and \texttt{git\ commit}.
\end{enumerate}

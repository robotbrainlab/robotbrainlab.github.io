  \begin{tikzpicture}[x=1pt, y=1pt, calloutline]
    \draw (0,0) circle (7);                                   % bulb

    \draw (-3.2,-9.5) -- (-3.2,-12.8) -- (3.2,-12.8) -- (3.2,-9.5); % base
